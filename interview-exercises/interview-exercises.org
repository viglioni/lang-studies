* Sorting input
Given a search string and a set of unsorted results, sort search results in the following order:
1. exact match
2. one of the words is exact match
3. starts with
4. rest, alphanumerically

#+begin_src elixir :exports both :results output
  
  result_list = ["Word", "Microsoft Word", "Wordpress", "1Password", "Google AdWords"]

  searchInput = "word"

  # solution

  sort_results = fn input, result_list ->
    search = String.downcase(input)

    result_list
    |> Enum.sort()                       # 4. alphanum
    |> Enum.sort_by(                     # 3. starts with
      fn str ->
        str
        |> String.downcase()
        |> String.starts_with?(search)
      end,
      :desc
    )
    |> Enum.sort_by(                     # 2. matches a word
      fn str ->
        str
        |> String.downcase()
        |> String.split(" ")
        |> Enum.member?(search)
      end,
      :desc
    )
    |> Enum.sort_by(                    # 1. perfect match
      fn str -> String.downcase(str) == search end,
    :desc)
  end

  sort_results.(searchInput, result_list)
#+end_src

#+begin_src js :exports both :results output 
  const resultList = ["Word", "Microsoft Word", "Wordpress", "1Password", "Google AdWords"]


  // solution
  const sortResults = (resultList, searchInput) => {
    const search = searchInput.toLowerCase()
    return  resultList
      .sort() // 4. alpha
      .sort((str1, str2) => { // 3. starts with
        if(str1.toLowerCase().startsWith(search)) return -1
        if(str1.toLowerCase().startsWith(search)) return 1
        return 0
      })
      .sort((str1, str2) => { // 2. word match
        if(str1.toLowerCase().split(" ").includes(search)) return -1
        if(str2.toLowerCase().split(" ").includes(search)) return 1
        return 0
      })
      .sort((str1, str2) => { // 1. perfect match
        if(str1.toLowerCase() === search) return -1
        if(str2.toLowerCase() === search) return 1
        return 0     
      })
  }

  console.log(sortResults(resultList, "word"))

#+end_src

* 2. How to find the duplicate number on a given integer array?

  repeated = Enum.random(1..1000)

  arr =  [repeated | Enum.to_list(1..1000 )]

  # solution without using Enum feats except for reduce
  # returning an array covers the case with multiple repetitions
  sol_2 = fn arr ->
    arr
    |> Enum.reduce(
      %{},
    fn val, acc -> Map.put(acc, val, 1 + Map.get(acc, val, 0)) end

    |> Map.keys()
  end

  sol_2.(arr)

* 3. How to find the largest and smallest number in an unsorted integer array?

#+begin_src elixir :exports both :results output 
  arr = 1..10 |> Enum.map(fn _ -> Enum.random(-100..100) end)

  # solution without using min/max functions

  sol_3 = fn list ->
    list
    |> Enum.reduce(
      %{},
    fn val, acc ->
      curr_min = Map.get(acc, :min, val)
      min = if val < curr_min do val else curr_min end
      curr_max = Map.get(acc, :max, val)
      max = if val > curr_max do val else curr_max end      
      %{min: min, max: max}
    end
    )
  end

  IO.puts(Enum.max(arr))
  IO.puts(Enum.min(arr))
  sol_3.(arr)

* 4. How to find all pairs of integer arrays whose sum is equal to a given number?

#+begin_src elixir :exports both :results output 
  arr = 1..1000 |> Enum.map(fn _ -> Enum.random(0..1000) end)
  sum_number = 1968

  # solution
  sol_4 = fn list, sum_val ->
    list
    |> Enum.flat_map(fn x ->
      list
      |> Enum.filter(fn y -> x + y == sum_val end)
      |> Enum.map(fn y -> Enum.sort([x,y]) end)
    end)
    |> Enum.filter(fn el -> !Enum.empty?(el) end)
    |> Enum.uniq()
  end

  sol_4.(arr, sum_number)
#+end_src

* 19. How to find all permutations of String?

#+begin_src elixir :exports both :results output 
  count_letters = fn str ->
    str
    |> String.to_charlist()
    |> Enum.reduce(
      %{},
    fn val, acc ->
      Map.put(acc, val, 1 + Map.get(acc, val, 0))
    end
    )
  end

  apply_mult = fn lst -> Enum.reduce(lst, 1, &Kernel.*/2)  end
  factorial = fn n -> (1..n) |> apply_mult.() end

  sol_19 = fn str ->
    # perm = n!/(m!p!..) where n is length,
    # and m,p,... is the number of repetitions of each letter

    repetitions = str
    |> count_letters.()
    |> Map.values()
    |> Enum.map(&(factorial.(&1)))
    |> apply_mult.()

    (str |> String.length() |> factorial.())/repetitions
  end

  sol_19.("banana")

#+end_src

#+RESULTS:
: 60.0

  const printNode = node => node
    ? console.log( {value: node.value, next: node.next?.value || null})
    : console.log(node)

  const printList = (node) => {
    let pointer = node
    while(pointer.next != null){
      printNode(pointer)
      pointer = pointer.next
    }
  }

  // messing-with-pointers solution

  const reverseList = (headNode) => {
    const stack = []
    let pointer = headNode

    while(pointer?.next != null){
      stack.push(pointer)
      pointer = pointer.next    
    }

    const newFirstNode = pointer

    while(stack.length) {
      pointer.next = stack.pop()
      pointer = pointer.next      
    }
    pointer.next = null

    return newFirstNode
  }


  const oldLastNode = reverseList(headNode)

  printList(oldLastNode)


  // functional solution, reversing it back

  const duplicate = el => el && JSON.parse(JSON.stringify(el))

  const linkFromStack = (stack,  newHeadNode = null) => {
    const [node , nextNode, ...rest] = stack


    // first execution, node is the last element of the stack
    if(stack[0].next == null) {      
      node.next = nextNode      
      return linkFromStack( [nextNode, ...rest], {...node}) 
    }


    // last execution:
    if(stack.length == 1) {
      node.next = null 
      return newHeadNode
    }

    // other cases
    node.next = nextNode 
    return linkFromStack([nextNode, ...rest],  newHeadNode)

  }

  const recReverse = (node, stack = []) =>
    node
    ? recReverse(node.next, [node, ...stack])
    : linkFromStack(duplicate(stack)) 


  console.log("\n============ using functional")
  printList(recReverse(oldLastNode))
  console.log("=========\ndid not change original list\n==========\n")
  printList(oldLastNode)
